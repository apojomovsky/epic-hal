# Capacitive Sensing (CPS) for pic16f193x-hal Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

Status: **implemented and cleared the §4 gate.** Peripheral #11 in
`docs/pic16f193x-plan.md` §7's roadmap table.

**Goal:** Land a `EPIC_CPS_*` driver for the Capacitive Sensing module
(DS41364B §18.0) through the §4 gate.

**Architecture:** CPS drives a small current through a selected pin
and routes its oscillation into Timer0's clock input (`T0XCS` bit) for
period measurement; it has no counter or digital output of its own.
Two registers: `CPSCON0` (enable, range, raw output bit, Timer0
routing) and `CPSCON1` (channel select). The driver's job is
enable/configure only; measurement is the caller's responsibility via
the **already-implemented** `pic16f193x_timer0` driver
(`pic16f193x-hal/include/peripherals/pic16f193x_timer0.h`, read it
before Task 3). No existing analog in `pic16f87xa-hal`/`pic18fxx5x-hal`.

**Tech Stack:** C99, MPLAB XC8, gcc (host sim), CMake, GNU Make.

## Global Constraints

- C99, no compiler extensions. No em-dashes anywhere.
- Every register access is a compile-time-constant `PIC_REG_*` token.
- Handle-copy convention: copy by value, never store the pointer.
- **Standing warning**: every bit position here was re-derived from
  DS41364B and cross-checked against the installed DFP header's
  `_CPSCON0_<FIELD>_POSN`/`_MASK` and `_CPSCON1_<FIELD>_POSN`/`_MASK`
  macros
  (`/opt/microchip/xc8/v3.10/pic/packs/Microchip.PIC12-16F1xxx_DFP/xc8/pic/include/proc/pic16f1937.h`).
  Timer1 shipped with a wrong bit position (`PIC_T1CON_TMR1CS` at bit
  1, copied from `pic16f87xa`'s classic-PIC16 T1CON, real field is
  bits 7:6) from a transcription without this cross-check. Never trust
  a bit position from another family's driver or memory alone. No
  existing CPS driver anywhere in this codebase to copy from.

## Non-goals (not in this phase)

- Real touch/proximity detection logic (debounce, threshold
  calibration): application-layer concern, out of scope for this HAL,
  which only exposes register control.
- Channels beyond what's bonded out on the target package: gated by a
  per-device capability check the driver validates against, not a
  hardcoded assumption of all 16.

```c
/**
 * @file    pic16f193x_cps.h
 * @brief   PIC16F193X Capacitive Sensing driver. Enable/configure
 *          only; period measurement is the caller's job via the
 *          existing EPIC_TIMER0_* driver once T0XCS routes CPS into it.
 * @details Source: DS41364B Register 18-1/18-2, DS41364B §18.0. Full
 *          reference: MANUAL.md §22.
 */
#ifndef PIC16F193X_CPS_H
#define PIC16F193X_CPS_H

#include <stdint.h>
#include "core/hal_status.h"

#ifdef __cplusplus
extern "C" {
#endif

typedef enum {
    CPS_RANGE_LOW    = 0x0U,
    CPS_RANGE_MEDLOW = 0x1U,
    CPS_RANGE_MEDHI  = 0x2U,
    CPS_RANGE_HIGH   = 0x3U,
} CPS_RangeTypeDef;

typedef struct {
    uint8_t Channel;          /**< CPSCH<3:0>, 0-15; validated against the device's bonded-out count. */
    CPS_RangeTypeDef Range;   /**< CPSRNG<1:0> oscillator current range. */
    uint8_t RouteToTimer0;    /**< Nonzero: set T0XCS so Timer0 clocks off the CPS oscillator. */
} CPS_HandleTypeDef;

#define CPS_HANDLE_DEFAULT { .Channel = 0U, .Range = CPS_RANGE_LOW, .RouteToTimer0 = 0U }

EPIC_StatusTypeDef EPIC_CPS_Init(const CPS_HandleTypeDef *h);
EPIC_StatusTypeDef EPIC_CPS_DeInit(void);
EPIC_StatusTypeDef EPIC_CPS_Enable(void);
EPIC_StatusTypeDef EPIC_CPS_Disable(void);
EPIC_StatusTypeDef EPIC_CPS_SelectChannel(uint8_t channel);
uint8_t EPIC_CPS_ReadRawOutput(void);   /**< Reads CPSOUT directly. */

#ifdef __cplusplus
}
#endif
#endif /* PIC16F193X_CPS_H */
```

**Files:** Create `pic16f193x-hal/src/peripherals/pic16f193x_cps.c`.

- [ ] **Step 1: Read `pic16f193x_timer0.h`** to confirm the exact API

- [ ] **Step 2: Write the driver body**

```c
/**
 * @file    pic16f193x_cps.c
 * @brief   PIC16F193X Capacitive Sensing driver implementation.
 */
#include "peripherals/pic16f193x_cps.h"
#include "pic16f193x_sfr.h"
#include "pic16f193x_platform.h"
#include "pic16f193x.h"   /* Per-device channel-count capability macro, e.g. PIC16F193X_FAMILY_CPS_CHANNELS. */

EPIC_StatusTypeDef EPIC_CPS_Init(const CPS_HandleTypeDef *h)
{
    if (!h) return EPIC_INVALID;
    if (h->Channel > 0x0FU) return EPIC_INVALID;
    /* Confirm PIC16F193X_FAMILY_CPS_CHANNELS (or equivalent capability
     * macro) exists in pic16f193x.h before relying on it here; if it
     * doesn't exist yet, add it there first, following the same
     * pattern as ADC_CH, rather than skipping the bounds check. */
    if (h->Channel >= PIC16F193X_FAMILY_CPS_CHANNELS) return EPIC_INVALID;

    EPIC_REG8(PIC_REG_CPSCON1) = (uint8_t)(h->Channel & PIC_CPSCON1_CPSCH_MASK);

    uint8_t con0 = (uint8_t)(((uint8_t)h->Range << 2) & PIC_CPSCON0_CPSRNG_MASK);
    if (h->RouteToTimer0) con0 |= PIC_CPSCON0_T0XCS;
    EPIC_REG8(PIC_REG_CPSCON0) = con0;   /* CPSON left clear until Enable(). */

    return EPIC_OK;
}

EPIC_StatusTypeDef EPIC_CPS_DeInit(void)
{
    EPIC_REG8(PIC_REG_CPSCON0) = 0x00U;
    EPIC_REG8(PIC_REG_CPSCON1) = 0x00U;
    return EPIC_OK;
}

EPIC_StatusTypeDef EPIC_CPS_Enable(void)
{
    EPIC_BIT_SET(EPIC_REG8(PIC_REG_CPSCON0), PIC_CPSCON0_CPSON);
    return EPIC_OK;
}

EPIC_StatusTypeDef EPIC_CPS_Disable(void)
{
    EPIC_BIT_CLR(EPIC_REG8(PIC_REG_CPSCON0), PIC_CPSCON0_CPSON);
    return EPIC_OK;
}

EPIC_StatusTypeDef EPIC_CPS_SelectChannel(uint8_t channel)
{
    if (channel >= PIC16F193X_FAMILY_CPS_CHANNELS) return EPIC_INVALID;
    EPIC_REG8(PIC_REG_CPSCON1) = (uint8_t)(channel & PIC_CPSCON1_CPSCH_MASK);
    return EPIC_OK;
}

uint8_t EPIC_CPS_ReadRawOutput(void)
{
    return (EPIC_REG8(PIC_REG_CPSCON0) & PIC_CPSCON0_CPSOUT) ? 1U : 0U;
}
```

If `PIC16F193X_FAMILY_CPS_CHANNELS` doesn't exist in `pic16f193x.h`
yet, add it there as part of this task (per-device value, following
the existing `ADC_CH`-style capability-macro pattern), don't invent a
local constant that bypasses the established per-device mechanism.
Confirm the exact platform macro names (`EPIC_REG8`/`EPIC_BIT_SET`/
`_CLR`) against a fresh read of `pic16f193x_timer1.c` before
finalizing.

```c
static uint8_t s_cps_injected_out;  /* Test-injected via
                                        pic16f193x_sim_cps_drive(),

                                        comparator injection-hook
                                        naming convention, check
                                        pic16f193x_sim.c for it. */

void pic16f193x_sim_cps_drive(uint8_t out)
{
    s_cps_injected_out = out ? 1U : 0U;
}

static void sim_step_cps(void)
{
    uint8_t con0 = pic16f193x_sim_sfr[PIC_REG_CPSCON0];
    if (!(con0 & PIC_CPSCON0_CPSON)) return;
    if (s_cps_injected_out) {
        pic16f193x_sim_sfr[PIC_REG_CPSCON0] = (uint8_t)(con0 | PIC_CPSCON0_CPSOUT);
    } else {
        pic16f193x_sim_sfr[PIC_REG_CPSCON0] = (uint8_t)(con0 & (uint8_t)~PIC_CPSCON0_CPSOUT);
    }
    /* T0XCS routing itself needs no extra sim work: Timer0's existing
     * sim_step_timer0() already free-runs off whatever clock source
     * is selected via the shared TMR0 control bits; confirm during
     * implementation that sim_step_timer0() reads T0XCS (or the
     * option-register T0CS bit it already models) correctly for this
     * routing, extending it only if it doesn't already handle an
     * external-clock-selected state generically. */
}
```
