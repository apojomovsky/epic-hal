# FVR for pic16f193x-hal Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

Status: **implemented and cleared the §4 gate.** Peripheral #9 in
`docs/pic16f193x-plan.md` §7's roadmap table. Register facts were
cross-checked against the installed DFP header before writing (Global
Constraints), not a substitute for the §4 gate actually running.

**Goal:** Land a `EPIC_FVR_*` driver for the Fixed Voltage Reference
(DS41364B §20.0) through the §4 gate.

**Architecture:** One register, `FVRCON`. Two independent gain-select
buffers (ADC-facing `ADFVR`, comparator/DAC-facing `CDAFVR`), a master
enable (`FVREN`), and a read-only ready flag (`FVRRDY`) that must be
polled before the output is trusted. Temperature-indicator bits
(`TSEN`/`TSRNG`) share this register but are a logically separate
feature, deferred (Non-goals). Mirrors `pic16f193x_timer1.{h,c}`'s
file layout and handle-copy convention; no existing analog in
`pic16f87xa-hal`/`pic18fxx5x-hal` to cross-check API shape against.

**Tech Stack:** C99, MPLAB XC8, gcc (host sim), CMake, GNU Make.

## Global Constraints

- C99, no compiler extensions in portable code.
- No em-dashes anywhere.
- Every register access is a compile-time-constant `PIC_REG_*` token.
- Handle-copy convention: copy by value, never store the pointer.
- **Standing warning**: every bit position here was re-derived from
  DS41364B and cross-checked against the installed DFP header's
  `_FVRCON_<FIELD>_POSN`/`_MASK` macros
  (`/opt/microchip/xc8/v3.10/pic/packs/Microchip.PIC12-16F1xxx_DFP/xc8/pic/include/proc/pic16f1937.h`).
  Timer1 shipped with a wrong bit position (`PIC_T1CON_TMR1CS` at bit
  1, copied from `pic16f87xa`'s classic-PIC16 T1CON, real field is
  bits 7:6) from a transcription without this cross-check. Never trust
  a bit position from another family's driver or memory alone. No
  existing FVR driver anywhere in this codebase to copy from.

## Non-goals (not in this phase)

- Temperature indicator (`TSEN`/`TSRNG`): a logically separate
  feature sharing this register; its own small follow-up plan, not
  because it's hard, because it's a different feature.
- This peripheral is a dependency FOR the ADC plan's `ADFVR`-as-source
  option and the DAC plan's `CDAFVR`-as-`DACPSS`-source option; both of
  those plans already scope FVR-as-reference out of their own first
  pass, so landing order is flexible, this plan does not block on
  either of them.

```c
 * installed DFP header (pic16f1937.h) _FVRCON_* POSN/MASK macros;
 * re-verify against DS41364B before relying on them. */
#define PIC_REG_FVRCON          0x117U

#define PIC_FVRCON_ADFVR_MASK   0x03U        /* FVRCON<1:0>: ADC-buffer gain. */
#define PIC_FVRCON_ADFVR_OFF    0x00U
#define PIC_FVRCON_ADFVR_1X     0x01U

#define PIC_FVRCON_TSRNG        PIC8_BIT(4)  /* FVRCON<4>: temp-indicator range. */
#define PIC_FVRCON_TSEN         PIC8_BIT(5)  /* FVRCON<5>: temp-indicator enable. */
#define PIC_FVRCON_FVRRDY       PIC8_BIT(6)  /* FVRCON<6>: ready flag, read-only. */
#define PIC_FVRCON_FVREN        PIC8_BIT(7)  /* FVRCON<7>: master enable. */
```

```c
/**
 * @file    pic16f193x_fvr.h
 * @brief   PIC16F193X Fixed Voltage Reference driver.
 * @details Source: DS41364B Register 20-1, DS41364B §20.0. Full
 *          reference: MANUAL.md §20.
 */
#ifndef PIC16F193X_FVR_H
#define PIC16F193X_FVR_H

#include <stdint.h>
#include "core/hal_status.h"

#ifdef __cplusplus
extern "C" {
#endif

/** FVR buffer gain, shared encoding for both the ADC buffer and the
 *  comparator/DAC buffer (independently selectable). */
typedef enum {
    FVR_GAIN_OFF = 0x0U,
    FVR_GAIN_1X  = 0x1U,
    FVR_GAIN_2X  = 0x2U,
    FVR_GAIN_4X  = 0x3U,
} FVR_GainTypeDef;

typedef struct {
    FVR_GainTypeDef ADCGain;      /**< ADFVR<1:0> initial value. */
    FVR_GainTypeDef CompDACGain;  /**< CDAFVR<1:0> initial value. */
} FVR_HandleTypeDef;

#define FVR_HANDLE_DEFAULT { .ADCGain = FVR_GAIN_OFF, .CompDACGain = FVR_GAIN_OFF }

EPIC_StatusTypeDef EPIC_FVR_Init(const FVR_HandleTypeDef *h);
EPIC_StatusTypeDef EPIC_FVR_DeInit(void);
EPIC_StatusTypeDef EPIC_FVR_Enable(void);
EPIC_StatusTypeDef EPIC_FVR_Disable(void);
EPIC_StatusTypeDef EPIC_FVR_SetADCGain(FVR_GainTypeDef gain);
EPIC_StatusTypeDef EPIC_FVR_SetCompDACGain(FVR_GainTypeDef gain);
uint8_t EPIC_FVR_IsReady(void);   /**< Polls FVRRDY; 1 = ready, 0 = not yet. */

#ifdef __cplusplus
}
#endif
#endif /* PIC16F193X_FVR_H */
```

- [ ] **Step 1: Write the driver body**

```c
/**
 * @file    pic16f193x_fvr.c
 * @brief   PIC16F193X FVR driver implementation.
 */
#include "peripherals/pic16f193x_fvr.h"
#include "pic16f193x_sfr.h"
#include "pic16f193x_platform.h"

EPIC_StatusTypeDef EPIC_FVR_Init(const FVR_HandleTypeDef *h)
{
    if (!h) return EPIC_INVALID;

    uint8_t con = (uint8_t)(h->ADCGain & PIC_FVRCON_ADFVR_MASK);
    con |= (uint8_t)((h->CompDACGain << PIC_FVRCON_CDAFVR_SHIFT) & PIC_FVRCON_CDAFVR_MASK);
    PIC8_REG8(PIC_REG_FVRCON) = con;   /* FVREN left clear until Enable(). */

    return EPIC_OK;
}

EPIC_StatusTypeDef EPIC_FVR_DeInit(void)
{
    PIC8_REG8(PIC_REG_FVRCON) = 0x00U;
    return EPIC_OK;
}

EPIC_StatusTypeDef EPIC_FVR_Enable(void)
{
    PIC8_BIT_SET(PIC8_REG8(PIC_REG_FVRCON), PIC_FVRCON_FVREN);
    return EPIC_OK;
}

EPIC_StatusTypeDef EPIC_FVR_Disable(void)
{
    PIC8_BIT_CLR(PIC8_REG8(PIC_REG_FVRCON), PIC_FVRCON_FVREN);
    return EPIC_OK;
}

EPIC_StatusTypeDef EPIC_FVR_SetADCGain(FVR_GainTypeDef gain)
{
    uint8_t con = PIC8_REG8(PIC_REG_FVRCON);
    con = (uint8_t)((con & (uint8_t)~PIC_FVRCON_ADFVR_MASK) | ((uint8_t)gain & PIC_FVRCON_ADFVR_MASK));
    PIC8_REG8(PIC_REG_FVRCON) = con;
    return EPIC_OK;
}

EPIC_StatusTypeDef EPIC_FVR_SetCompDACGain(FVR_GainTypeDef gain)
{
    uint8_t con = PIC8_REG8(PIC_REG_FVRCON);
    uint8_t shifted = (uint8_t)(((uint8_t)gain << PIC_FVRCON_CDAFVR_SHIFT) & PIC_FVRCON_CDAFVR_MASK);
    con = (uint8_t)((con & (uint8_t)~PIC_FVRCON_CDAFVR_MASK) | shifted);
    PIC8_REG8(PIC_REG_FVRCON) = con;
    return EPIC_OK;
}

uint8_t EPIC_FVR_IsReady(void)
{
    return (PIC8_REG8(PIC_REG_FVRCON) & PIC_FVRCON_FVRRDY) ? 1U : 0U;
}
```

- [ ] **Step 1: Add the sim step**

```c
static void sim_step_fvr(void)
{
    /* No real analog settling time to simulate: FVRRDY goes ready on
     * the sim step immediately after FVREN is set. This is a known
     * simplification, documented here rather than left implicit. */
    uint8_t con = pic16f193x_sim_sfr[PIC_REG_FVRCON];
    if (con & PIC_FVRCON_FVREN) {
        pic16f193x_sim_sfr[PIC_REG_FVRCON] = (uint8_t)(con | PIC_FVRCON_FVRRDY);
    } else {
        pic16f193x_sim_sfr[PIC_REG_FVRCON] = (uint8_t)(con & (uint8_t)~PIC_FVRCON_FVRRDY);
    }
}
```

## Task 6: Add `example_fvr.c`

**Files:** Create `pic16f193x-hal/tests/example_fvr.c`.

**Interfaces:** Consumes `EPIC_FVR_Init/Enable/IsReady` (Task 2/3),

- [ ] **Step 1: Write the example**

```c
/**
 * @file    example_fvr.c
 * @brief   FVR smoke test: enable with 2x ADC gain, poll ready, report.
 *
 * @details
 *   Expected register image (host sim, after init and one sim step):
 *     FVRCON = 0xC2   (FVREN=1, FVRRDY=1 after the sim step,
 *                      TSEN=0, CDAFVR=00, ADFVR=10 (2x))
 *     TRISA  = 0xFE, LATA = 0x00 (harness marker pin, see example_timer1.c)
 */

#include "pic16f193x.h"
#include "pic16f193x_sfr.h"
#include "peripherals/pic16f193x_gpio.h"
#include "peripherals/pic16f193x_fvr.h"
#include "core/pic16f193x_wdt_sleep.h"
#include "core/pic8_harness.h"

extern void pic16f193x_harness_halt(void);

int main(void)
{
    pic8_harness_init();

    FVR_HandleTypeDef fvr = FVR_HANDLE_DEFAULT;
    fvr.ADCGain = FVR_GAIN_2X;
    EPIC_FVR_Init(&fvr);
    EPIC_FVR_Enable();

    /* On host sim, one call into whatever advances the sim by a step
     * is needed here before FVRRDY goes high; use the same
     * step-pumping mechanism example_timer1.c's main loop uses (check
     * that file for the exact call, likely something the harness does
     * implicitly via pic8_harness_report or an explicit sim-step call
     * on host builds only). On real target there is no sim to step;
     * EPIC_FVR_IsReady() polls the real FVRRDY bit directly, spin until
     * ready or a bounded retry count, don't spin forever unconditionally. */
    uint8_t ready = 0U;
    for (int i = 0; i < 1000 && !ready; i++) {
        ready = EPIC_FVR_IsReady();
    }

    uint8_t con = PIC8_REG8(PIC_REG_FVRCON);
    pic8_harness_log("FVRCON=0x%02X ready=%u\n", con, (unsigned)ready);
    int rc = pic8_harness_report(ready && (con == 0xC2U));
    pic16f193x_harness_halt();
    return rc;
}
```
