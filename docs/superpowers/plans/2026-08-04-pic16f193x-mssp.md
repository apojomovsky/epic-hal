Status: **implemented and cleared the §4 gate.** Peripheral #4 in
`docs/pic16f193x-plan.md` §7's roadmap table. SPI master only (I2C
deferred). Host sim passes (SSPSTAT=0x40, SSPCON1=0x20, PASS). 6-part
XC8 build all clean. mdb MODE=gpio gate PASSes (PORTA=1).

**Goal:** Land a `HAL_SSP_*` driver for MSSP in **SPI master mode
only** (DS41364B §17.0 in this datasheet's chapter numbering for MSSP;
confirm exact section number during implementation) through the §4
gate.

**Architecture:** Mirrors `pic16f87xa_ssp.{h,c}`'s Cube-style handle
and API, scoped down: only the `SSP_MODE_SPI_MASTER_*` encodings are
accepted by `HAL_SSP_Init`, everything I2C (`SSPCON2`/`SSPCON3`,
Start/Stop/ReceiveEnable/Acknowledge) is entirely absent from this
driver, not stubbed-and-rejected (I2C is a substantially different,
larger scope, its own follow-up plan, not squeezed in here). SPI
master is self-contained: no bus arbitration, no clock stretching, no
multi-master concerns, the natural minimal-viable first cut, same
reasoning Timer1 used for internal-clock-only.

**Tech Stack:** C99, MPLAB XC8, gcc (host sim), CMake, GNU Make.

## Global Constraints

- C99, no compiler extensions. No em-dashes anywhere.
- Every register access is a compile-time-constant `PIC_REG_*` token.
- Handle-copy convention: copy by value, never store the pointer.
- **Standing warning**: every bit position here was re-derived from
  DS41364B and cross-checked against the installed DFP header's
  `_SSPSTAT_`/`_SSPCON1_` `_POSN`/`_MASK` macros
  (`/opt/microchip/xc8/v3.10/pic/packs/Microchip.PIC12-16F1xxx_DFP/xc8/pic/include/proc/pic16f1937.h`).
  Timer1 shipped with a wrong bit position (`PIC_T1CON_TMR1CS` at bit
  1, copied from `pic16f87xa`'s classic-PIC16 T1CON, real field is
  bits 7:6) from a transcription without this cross-check. Never trust
  a bit position from another family's driver or memory alone. Note
  this family's `SSPSTAT` has `R_nW`/`D_nA` at bits 2/5 where
  `pic16f87xa`'s equivalent register may number them differently;
  don't assume identical bit numbering between families for any
  register, confirmed here via direct DFP header grep, not carried
  over from the reference driver.
- **Errata (DS80000479, `docs/pic16f193x-plan.md` §1)**: "MSSP SPI
  master BF/SSPIF set half SCK early (CKE=0)". This driver's default
  handle uses `SSP_SPI_CKE_IDLE_ACTIVE` (CKE=1), away from the
  erratum-affected CKE=0 mode; `HAL_SSP_Init` does not forbid CKE=0
  (some SPI peripherals genuinely require it for protocol
  compatibility) but the header/MANUAL must document the erratum
  explicitly wherever `ClockEdge` is documented, so a caller who picks
  CKE=0 does so knowingly.

## Non-goals (not in this phase)

- All I2C modes (master/slave/firmware-controlled): a substantially
  different, larger scope (bus protocol state machine, clock
  stretching, addressing), a follow-up plan of its own.
- SPI slave mode: master-only this phase, for the same
  self-containment reasoning as the Architecture section states.
- `SSPMSK` (I2C address masking): irrelevant without I2C slave mode.

- [ ] **Step 1: Add `SSPSTAT`/`SSPCON1` bit macros** (addresses

```c
/* Bank 4. SSPSTAT/SSPCON1 (SPI-relevant subset only, I2C-only fields
 * on SSPCON2/SSPCON3 not defined this phase). DS41364B MSSP chapter.
 * Verified against the installed DFP header (pic16f1937.h) _SSPSTAT_/
 * _SSPCON1_ POSN/MASK macros; re-verify against DS41364B before
 * relying on them. */
#define PIC_SSPSTAT_BF          PIC8_BIT(0)   /* Buffer full, read-only. */
#define PIC_SSPSTAT_R_NW        PIC8_BIT(2)   /* I2C-only, unused this phase. */
#define PIC_SSPSTAT_CKE         PIC8_BIT(6)   /* SPI clock edge select. */
#define PIC_SSPSTAT_SMP         PIC8_BIT(7)   /* SPI sample phase, master only. */
/* SSPSTAT POR = 0x00 (confirm against DS41364B). */

#define PIC_SSPCON1_SSPM_MASK   0x0FU         /* Mode select, SPI-master encodings 0000-0011 this phase. */
#define PIC_SSPCON1_CKP         PIC8_BIT(4)   /* Clock polarity. */
#define PIC_SSPCON1_SSPEN       PIC8_BIT(5)   /* Module enable. */
#define PIC_SSPCON1_SSPOV       PIC8_BIT(6)   /* Receive overflow, I2C-relevant but readable in SPI too per datasheet, confirm. */
#define PIC_SSPCON1_WCOL        PIC8_BIT(7)   /* Write collision. */
/* SSPCON1 POR = 0x00 (confirm against DS41364B). */
```

```c
/**
 * @file    pic16f193x_ssp.h
 * @brief   PIC16F193X MSSP driver, SPI master mode only this phase
 *          (I2C and SPI slave are a follow-up, see the plan's
 *          Non-goals). CKE=0 carries a real DS80000479 errata
 *          (BF/SSPIF set half SCK early); the default handle uses
 *          CKE=1, away from it, but CKE=0 is not forbidden.
 * @details Source: DS41364B MSSP chapter. API shape mirrors
 *          pic16f87xa_ssp.h's SPI-relevant subset. Full reference:
 *          MANUAL.md §15.
 */
#ifndef PIC16F193X_SSP_H
#define PIC16F193X_SSP_H

#include <stdint.h>
#include "core/hal_status.h"

#ifdef __cplusplus
extern "C" {
#endif

/** SSPCON1<SSPM3:0>, SPI-master encodings only this phase. */
typedef enum {
    SSP_MODE_SPI_MASTER_FOSC_4  = 0x0U,
    SSP_MODE_SPI_MASTER_FOSC_16 = 0x1U,
    SSP_MODE_SPI_MASTER_FOSC_64 = 0x2U,
    SSP_MODE_SPI_MASTER_TMR2    = 0x3U,   /**< Depends on the Timer2/4/6 plan if used; confirm dependency during implementation. */
} SSP_ModeTypeDef;

typedef enum {
    SSP_SPI_CKE_ACTIVE_IDLE = 0x0U,   /**< CKE=0: transmit on idle->active. DS80000479 errata applies, see file header. */
    SSP_SPI_CKE_IDLE_ACTIVE = 0x1U,   /**< CKE=1: transmit on active->idle. Errata-safe default. */
} SSP_ClockEdgeTypeDef;

typedef enum {
    SSP_SPI_CKP_IDLE_LOW  = 0x0U,
    SSP_SPI_CKP_IDLE_HIGH = 0x1U,
} SSP_ClockPolarityTypeDef;

typedef enum {
    SSP_SPI_SMP_MIDDLE = 0x0U,
    SSP_SPI_SMP_END    = 0x1U,
} SSP_SamplePhaseTypeDef;

typedef struct {
    SSP_ModeTypeDef          Mode;
    SSP_ClockEdgeTypeDef     ClockEdge;
    SSP_ClockPolarityTypeDef ClockPolarity;
    SSP_SamplePhaseTypeDef   SamplePhase;
    void (*TransferCallback)(void);   /**< Fires on SSPIF; NULL = no callback. */
} SSP_HandleTypeDef;

#define SSP_HANDLE_DEFAULT { .Mode = SSP_MODE_SPI_MASTER_FOSC_4, .ClockEdge = SSP_SPI_CKE_IDLE_ACTIVE, .ClockPolarity = SSP_SPI_CKP_IDLE_LOW, .SamplePhase = SSP_SPI_SMP_MIDDLE, .TransferCallback = 0 }

HAL_StatusTypeDef HAL_SSP_Init(const SSP_HandleTypeDef *h);
HAL_StatusTypeDef HAL_SSP_DeInit(void);
uint16_t HAL_SSP_WriteByte(uint8_t data);   /**< 0xFFFF if WCOL, byte not written. */
uint8_t  HAL_SSP_ReadByte(void);
uint8_t  HAL_SSP_IsBufferFull(void);
uint8_t  HAL_SSP_HasWriteCollision(void);
void     HAL_SSP_ClearWriteCollision(void);

void SSP_IRQHandler(void) PIC8_WEAK;

#ifdef __cplusplus
}
#endif
#endif /* PIC16F193X_SSP_H */
```

```c
/**
 * @file    pic16f193x_ssp.c
 * @brief   PIC16F193X MSSP driver implementation, SPI master only.
 */
#include "peripherals/pic16f193x_ssp.h"
#include "pic16f193x_sfr.h"
#include "pic16f193x_platform.h"

static void (*s_transfer_cb)(void);

HAL_StatusTypeDef HAL_SSP_Init(const SSP_HandleTypeDef *h)
{
    if (!h) return HAL_INVALID;
    if ((uint8_t)h->Mode > 0x03U) return HAL_INVALID;   /* Only the 4 SPI-master encodings this phase. */

    uint8_t stat = 0U;
    if (h->ClockEdge == SSP_SPI_CKE_IDLE_ACTIVE) stat |= PIC_SSPSTAT_CKE;
    if (h->SamplePhase == SSP_SPI_SMP_END) stat |= PIC_SSPSTAT_SMP;
    PIC8_REG8(PIC_REG_SSPSTAT) = stat;

    uint8_t con1 = (uint8_t)h->Mode & PIC_SSPCON1_SSPM_MASK;
    if (h->ClockPolarity == SSP_SPI_CKP_IDLE_HIGH) con1 |= PIC_SSPCON1_CKP;
    con1 |= PIC_SSPCON1_SSPEN;
    PIC8_REG8(PIC_REG_SSPCON1) = con1;

    s_transfer_cb = h->TransferCallback;

    return HAL_OK;
}

HAL_StatusTypeDef HAL_SSP_DeInit(void)
{
    PIC8_REG8(PIC_REG_SSPCON1) = 0x00U;
    PIC8_REG8(PIC_REG_SSPSTAT) = 0x00U;
    s_transfer_cb = 0;
    return HAL_OK;
}

uint16_t HAL_SSP_WriteByte(uint8_t data)
{
    if (PIC8_REG8(PIC_REG_SSPCON1) & PIC_SSPCON1_WCOL) return 0xFFFFU;
    PIC8_REG8(PIC_REG_SSPBUF) = data;
    if (PIC8_REG8(PIC_REG_SSPCON1) & PIC_SSPCON1_WCOL) return 0xFFFFU;   /* Write itself may set WCOL if BF was still set. */
    return (uint16_t)data;
}

uint8_t HAL_SSP_ReadByte(void)
{
    return PIC8_REG8(PIC_REG_SSPBUF);   /* Reading clears BF. */
}

uint8_t HAL_SSP_IsBufferFull(void)
{
    return (PIC8_REG8(PIC_REG_SSPSTAT) & PIC_SSPSTAT_BF) ? 1U : 0U;
}

uint8_t HAL_SSP_HasWriteCollision(void)
{
    return (PIC8_REG8(PIC_REG_SSPCON1) & PIC_SSPCON1_WCOL) ? 1U : 0U;
}

void HAL_SSP_ClearWriteCollision(void)
{
    PIC8_BIT_CLR(PIC8_REG8(PIC_REG_SSPCON1), PIC_SSPCON1_WCOL);
}

void SSP_IRQHandler(void)
{
    if (s_transfer_cb) s_transfer_cb();
}
```
