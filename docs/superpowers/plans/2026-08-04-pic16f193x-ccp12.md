# CCP1/CCP2 for pic16f193x-hal Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

Status: **implemented and cleared the §4 gate.** Peripheral #2 in
`docs/pic16f193x-plan.md` §7's roadmap table. Host sim passes
(CCP1CON=0x08, CCPR1=0x0100, CCP2CON=0x08, CCPR2=0x0200, PASS). 6-part
XC8 build all clean. mdb MODE=gpio gate PASSes (PORTA=1). Manual
readback confirms all expected registers. One deviation from the plan:
the driver uses branch-before-touch macros (CCP_WRITE_CPRH etc.) instead
of the plan's instance_regs() helper, which returned runtime SFR
addresses via output parameters and violated the Global Constraint
(every register access is a compile-time-constant PIC_REG_* token).

**Goal:** Land a `HAL_CCP_*` driver for CCP1 and CCP2 (both Enhanced
CCP on this device, DS41364B §15.0) through the §4 gate, scoped to
capture and compare modes only this phase (Non-goals covers PWM).

**Architecture:** CCP1/CCP2 are register-identical (both full ECCP:
`CCPRxL/H`, `CCPxCON`, `PWMxCON`, `CCPxAS`, `PSTRxCON`), unlike PIC18
where only CCP1 is enhanced. One driver, two instances, mirroring
`pic18fxx5x_ccp.{h,c}`'s Cube-style handle shape (`CCP_InstanceTypeDef`,
`CCP_ModeTypeDef`, `CompareValue`, `EventCallback`) but with the
PWM/dead-band/auto-shutdown fields omitted from the handle this phase
(rejected at `Init`, not silently ignored), the same scope-cut pattern
Timer1 used for `TIMER1_CLOCK_EXTERNAL`.

**Tech Stack:** C99, MPLAB XC8, gcc (host sim), CMake, GNU Make.

## Global Constraints

- C99, no compiler extensions. No em-dashes anywhere.
- Every register access is a compile-time-constant `PIC_REG_*` token.
- Handle-copy convention: copy by value, never store the pointer.
- **Standing warning**: every bit position here was re-derived from
  DS41364B and cross-checked against the installed DFP header's
  `_CCP1CON_`/`_CCP2CON_`/`_PWM1CON_`/`_PWM2CON_`/`_CCP1AS_`/`_CCP2AS_`/
  `_PSTR1CON_`/`_PSTR2CON_`/`_CCPTMRS0_`/`_CCPTMRS1_` `_POSN`/`_MASK`
  macros
  (`/opt/microchip/xc8/v3.10/pic/packs/Microchip.PIC12-16F1xxx_DFP/xc8/pic/include/proc/pic16f1937.h`).
  Timer1 shipped with a wrong bit position (`PIC_T1CON_TMR1CS` at bit
  1, copied from `pic16f87xa`'s classic-PIC16 T1CON, real field is
  bits 7:6) from a transcription without this cross-check. Never trust
  a bit position from another family's driver or memory alone.

## Non-goals (not in this phase)

- Enhanced/multi-output PWM (`PxM` steering, `PWMxCON` delay,
  `CCPxAS` auto-shutdown, `PSTRxCON` steering): a materially larger
  feature surface, deferred with `HAL_CCP_Init` returning `HAL_INVALID`
  for `CCP_MODE_PWM`, mirroring Timer1's `TIMER1_CLOCK_EXTERNAL`
  rejection precedent. DS80000479's "ECCP 0%-duty direction-change and
  port-steering issues" errata applies to this deferred PWM work, not
  to this plan's capture/compare scope; carry the errata note forward
  to whoever picks up PWM.
- `CCPTMRS0`/`CCPTMRS1` (Timer2/4/6 PWM-timebase select): only
  meaningful for PWM mode, so out of scope alongside it. Bit macros are
  still defined in Task 1 since they cost nothing and CCP3/4/5's plan
  (`docs/superpowers/plans/2026-08-04-pic16f193x-ccp345.md`) needs
  them too.

```c
/**
 * @file    pic16f193x_ccp.h
 * @brief   PIC16F193X CCP1/CCP2 driver, capture and compare modes
 *          only this phase (PWM deferred, see the plan's Non-goals).
 * @details Source: DS41364B Register 15-1, DS41364B §15.0. Handle
 *          shape mirrors pic18fxx5x_ccp.h's Cube-style API, adapted:
 *          both instances are Enhanced CCP on this device (unlike
 *          PIC18 where only CCP1 is). Full reference: MANUAL.md §13.
 */
#ifndef PIC16F193X_CCP_H
#define PIC16F193X_CCP_H

#include <stdint.h>
#include "core/hal_status.h"

#ifdef __cplusplus
extern "C" {
#endif

typedef enum {
    CCP_INSTANCE_1 = 1,
    CCP_INSTANCE_2 = 2,
} CCP_InstanceTypeDef;

/** CCPxCON<3:0> mode select, capture/compare encodings only this
 *  phase (DS41364B Register 15-1); PWM (11xx) is rejected by
 *  HAL_CCP_Init, not silently accepted. */
typedef enum {
    CCP_MODE_OFF              = 0x0U,  /**< 0000, module disabled. */
    CCP_MODE_CAPTURE_FALLING  = 0x4U,  /**< 0100, every falling edge. */
    CCP_MODE_CAPTURE_RISING   = 0x5U,  /**< 0101, every rising edge. */
    CCP_MODE_CAPTURE_4TH      = 0x6U,  /**< 0110, every 4th rising. */
    CCP_MODE_CAPTURE_16TH     = 0x7U,  /**< 0111, every 16th rising. */
    CCP_MODE_COMPARE_SET      = 0x8U,  /**< 1000, set output on match. */
    CCP_MODE_COMPARE_CLEAR    = 0x9U,  /**< 1001, clear output on match. */
    CCP_MODE_COMPARE_SOFT_IF  = 0xAU,  /**< 1010, software interrupt only. */
    CCP_MODE_COMPARE_TRIGGER  = 0xBU,  /**< 1011, special event trigger. */
    CCP_MODE_PWM              = 0xCU,  /**< 11xx: rejected, see above. */
} CCP_ModeTypeDef;

typedef struct {
    CCP_InstanceTypeDef Instance;
    CCP_ModeTypeDef      Mode;
    uint16_t              CompareValue;   /**< 16-bit compare/capture value. */
    void (*EventCallback)(void);          /**< Fires on CCPxIF; NULL = no callback, IRQ still clears the flag. */
} CCP_HandleTypeDef;

#define CCP_HANDLE_DEFAULT { .Instance = CCP_INSTANCE_1, .Mode = CCP_MODE_OFF, .CompareValue = 0U, .EventCallback = 0 }

HAL_StatusTypeDef HAL_CCP_Init(const CCP_HandleTypeDef *h);
HAL_StatusTypeDef HAL_CCP_DeInit(CCP_InstanceTypeDef inst);
HAL_StatusTypeDef HAL_CCP_SetCompare(CCP_InstanceTypeDef inst, uint16_t value);
uint16_t HAL_CCP_GetCapture(CCP_InstanceTypeDef inst);

void CCP1_IRQHandler(void) PIC8_WEAK;
void CCP2_IRQHandler(void) PIC8_WEAK;

#ifdef __cplusplus
}
#endif
#endif /* PIC16F193X_CCP_H */
```

```c
/**
 * @file    pic16f193x_ccp.c
 * @brief   PIC16F193X CCP1/CCP2 driver implementation.
 */
#include "peripherals/pic16f193x_ccp.h"
#include "pic16f193x_sfr.h"
#include "pic16f193x_platform.h"

static void (*s_ccp1_cb)(void);
static void (*s_ccp2_cb)(void);

static HAL_StatusTypeDef instance_regs(CCP_InstanceTypeDef inst,
                                        uint16_t *con_addr, uint16_t *ccpr_l_addr)
{
    switch (inst) {
    case CCP_INSTANCE_1: *con_addr = PIC_REG_CCP1CON; *ccpr_l_addr = PIC_REG_CCPR1L; return HAL_OK;
    case CCP_INSTANCE_2: *con_addr = PIC_REG_CCP2CON; *ccpr_l_addr = PIC_REG_CCPR2L; return HAL_OK;
    default: return HAL_INVALID;
    }
}

HAL_StatusTypeDef HAL_CCP_Init(const CCP_HandleTypeDef *h)
{
    if (!h) return HAL_INVALID;
    if (h->Mode == CCP_MODE_PWM) return HAL_INVALID;  /* Not in this phase, see Non-goals. */

    uint16_t con_addr, ccpr_l_addr;
    if (instance_regs(h->Instance, &con_addr, &ccpr_l_addr) != HAL_OK) return HAL_INVALID;

    /* High byte first, then low: matches this family's established
     * atomic-write idiom (pic16f193x_timer1.c's HAL_TIMER1_WriteCounter). */
    PIC8_REG8(ccpr_l_addr + 1U) = (uint8_t)(h->CompareValue >> 8);
    PIC8_REG8(ccpr_l_addr)      = (uint8_t)(h->CompareValue & 0xFFU);
    PIC8_REG8(con_addr) = (uint8_t)((uint8_t)h->Mode & PIC_CCP1CON_CCPM_MASK);

    if (h->Instance == CCP_INSTANCE_1) s_ccp1_cb = h->EventCallback;
    else s_ccp2_cb = h->EventCallback;

    return HAL_OK;
}

HAL_StatusTypeDef HAL_CCP_DeInit(CCP_InstanceTypeDef inst)
{
    uint16_t con_addr, ccpr_l_addr;
    if (instance_regs(inst, &con_addr, &ccpr_l_addr) != HAL_OK) return HAL_INVALID;
    PIC8_REG8(con_addr) = 0x00U;
    if (inst == CCP_INSTANCE_1) s_ccp1_cb = 0; else s_ccp2_cb = 0;
    return HAL_OK;
}

HAL_StatusTypeDef HAL_CCP_SetCompare(CCP_InstanceTypeDef inst, uint16_t value)
{
    uint16_t con_addr, ccpr_l_addr;
    if (instance_regs(inst, &con_addr, &ccpr_l_addr) != HAL_OK) return HAL_INVALID;
    PIC8_REG8(ccpr_l_addr + 1U) = (uint8_t)(value >> 8);
    PIC8_REG8(ccpr_l_addr)      = (uint8_t)(value & 0xFFU);
    return HAL_OK;
}

uint16_t HAL_CCP_GetCapture(CCP_InstanceTypeDef inst)
{
    uint16_t con_addr, ccpr_l_addr;
    if (instance_regs(inst, &con_addr, &ccpr_l_addr) != HAL_OK) return 0U;
    /* High-low-high retry idiom (pic16f193x_timer1.c's
     * HAL_TIMER1_ReadCounter): guards against a rollover between the
     * two byte reads. */
    uint8_t hi1, lo, hi2;
    do {
        hi1 = PIC8_REG8(ccpr_l_addr + 1U);
        lo  = PIC8_REG8(ccpr_l_addr);
        hi2 = PIC8_REG8(ccpr_l_addr + 1U);
    } while (hi1 != hi2);
    return (uint16_t)(((uint16_t)hi1 << 8) | lo);
}

void CCP1_IRQHandler(void)
{
    if (s_ccp1_cb) s_ccp1_cb();
}

void CCP2_IRQHandler(void)
{
    if (s_ccp2_cb) s_ccp2_cb();
}
```
